为了帮你高效对比YOLOv5和YOLOv8在停车场景（车辆+车位检测）的训练效果，我整理了**统一训练流程+对比验证方案**，包含适配两类模型的训练代码、评估指标、结果对比方法，确保你能在15天项目周期内快速得出结论。

### 一、核心准备：统一数据集（保证对比公平）
基于之前的AI标注+人工校准后的数据集，整理为YOLO通用格式（v5/v8均兼容）：
```
parking_dataset/
├── images/
│   ├── train/  # 训练集（70%校准后的图片/视频帧）
│   ├── val/    # 验证集（20%）
│   └── test/   # 测试集（10%）
├── labels/     # 标注文件（与images目录结构一一对应）
│   ├── train/
│   ├── val/
│   └── test/
└── data.yaml   # 数据集配置文件（关键！统一类别和路径）
```

#### 2. 数据集划分技巧
- 用脚本自动划分（避免手动操作）：
```python
import os
import random
import shutil
from pathlib import Path

# 配置路径
img_dir = "parking_dataset/images"
label_dir = "parking_dataset/labels"
split_ratio = [0.7, 0.2, 0.1]  # 训练/验证/测试

# 创建目录
for split in ["train", "val", "test"]:
    Path(f"{img_dir}/{split}").mkdir(parents=True, exist_ok=True)
    Path(f"{label_dir}/{split}").mkdir(parents=True, exist_ok=True)

# 获取所有图片
img_files = [f for f in os.listdir(img_dir) if not os.path.isdir(os.path.join(img_dir, f))]
random.shuffle(img_files)

# 划分数据
train_num = int(len(img_files) * split_ratio[0])
val_num = int(len(img_files) * split_ratio[1])

train_files = img_files[:train_num]
val_files = img_files[train_num:train_num+val_num]
test_files = img_files[train_num+val_num:]

# 复制文件
def copy_files(files, split):
    for f in files:
        # 复制图片
        shutil.copy(os.path.join(img_dir, f), os.path.join(img_dir, split, f))
        # 复制标注文件
        label_file = os.path.splitext(f)[0] + ".txt"
        if os.path.exists(os.path.join(label_dir, label_file)):
            shutil.copy(os.path.join(label_dir, label_file), os.path.join(label_dir, split, label_file))

copy_files(train_files, "train")
copy_files(val_files, "val")
copy_files(test_files, "test")

print(f"数据集划分完成：训练集{len(train_files)}张，验证集{len(val_files)}张，测试集{len(test_files)}张")
```

#### 2. YOLOv8 训练代码（推荐：API更简洁）
```python
from ultralytics import YOLO
import os

# 配置参数（统一超参数，保证对比公平）
DATA_YAML = "parking_dataset/data.yaml"
EPOCHS = 50  # 训练轮数（停车场景50轮足够）
BATCH_SIZE = 16  # 根据显卡显存调整（8/16/32）
IMG_SIZE = 640  # 输入图片尺寸
DEVICE = "0"  # 显卡编号（CPU用"cpu"）

# 加载预训练模型（v8s：中等尺寸，平衡精度和速度）
model_v8 = YOLO("yolov8s.pt")

# 开始训练
results_v8 = model_v8.train(
    data=DATA_YAML,
    epochs=EPOCHS,
    batch=BATCH_SIZE,
    imgsz=IMG_SIZE,
    device=DEVICE,
    project="parking_train",
    name="yolov8s_parking",
    save=True,  # 保存模型
    val=True    # 训练中验证
)

# 验证模型（输出关键指标）
metrics_v8 = model_v8.val(data=DATA_YAML, device=DEVICE)
# 保存测试结果
with open("yolov8_metrics.txt", "w") as f:
    f.write(f"mAP@0.5: {metrics_v8.box.map50:.4f}\n")
    f.write(f"车辆AP@0.5: {metrics_v8.box.ap50[0]:.4f}\n")
    f.write(f"车位AP@0.5: {metrics_v8.box.ap50[1]:.4f}\n")
    f.write(f"Precision: {metrics_v8.box.mp:.4f}\n")
    f.write(f"Recall: {metrics_v8.box.mr:.4f}\n")

print("YOLOv8训练完成！关键指标：")
print(f"mAP@0.5: {metrics_v8.box.map50:.4f}")
print(f"车辆检测AP: {metrics_v8.box.ap50[0]:.4f}")
print(f"车位检测AP: {metrics_v8.box.ap50[1]:.4f}")
```

#### 3. YOLOv5 训练代码（兼容官方仓库）
```python
import subprocess
import os

# 配置参数（与YOLOv8完全一致）
DATA_YAML = "parking_dataset/data.yaml"
EPOCHS = 50
BATCH_SIZE = 16
IMG_SIZE = 640
DEVICE = "0"
YOLOV5_DIR = "yolov5"  # YOLOv5仓库路径

# 构建训练命令
train_cmd = [
    "python", f"{YOLOV5_DIR}/train.py",
    "--data", DATA_YAML,
    "--epochs", str(EPOCHS),
    "--batch-size", str(BATCH_SIZE),
    "--imgsz", str(IMG_SIZE),
    "--device", DEVICE,
    "--weights", "yolov5s.pt",  # 对应v8s的v5s模型
    "--project", "parking_train",
    "--name", "yolov5s_parking",
    "--save",
    "--val"
]

# 执行训练
subprocess.run(train_cmd, check=True)

# 验证模型（输出指标）
val_cmd = [
    "python", f"{YOLOV5_DIR}/val.py",
    "--data", DATA_YAML,
    "--weights", "parking_train/yolov5s_parking/weights/best.pt",
    "--device", DEVICE,
    "--imgsz", str(IMG_SIZE),
    "--save-json"
]
result = subprocess.run(val_cmd, capture_output=True, text=True)

# 提取关键指标（或直接查看val.txt文件）
with open("parking_train/yolov5s_parking/val.txt", "r") as f:
    val_lines = f.readlines()
    # 保存指标
    with open("yolov5_metrics.txt", "w") as out_f:
        out_f.writelines(val_lines[-5:])  # 保存最后5行关键指标

print("YOLOv5训练完成！指标已保存至yolov5_metrics.txt")
```

### 三、关键对比维度（停车场景重点关注）
| 对比维度       | 评估方法                                                                 | 停车场景关注点                     |
|----------------|--------------------------------------------------------------------------|------------------------------------|
| 核心精度       | mAP@0.5（平均精度）、AP@0.5（车辆/车位分别的精度）| 车位检测AP（易漏标/错标，优先级更高） |
| 检测速度       | 单张图片推理时间（ms）、FPS                                             | 视频实时性（≥15FPS即可满足停车监控） |
| 漏检率         | 测试集中未检测到的车辆/车位数量 ÷ 总标注数量                             | 遮挡车位/车辆的漏检率（停车场高频场景） |
| 误检率         | 错误检测的目标数量 ÷ 检测到的总目标数量                                 | 非车位区域被标为车位的比例         |
| 模型大小       | 权重文件大小（MB）| 部署便捷性（后端/边缘设备适配）|

#### 快速对比脚本（自动整理指标）
```python

# 解析YOLOv8指标
with open("yolov8_metrics.txt", "r") as f:
    v8_lines = f.readlines()
v8_map50 = float(re.findall(r"\d+\.\d+", v8_lines[0])[0])
v8_car_ap = float(re.findall(r"\d+\.\d+", v8_lines[1])[0])
v8_parking_ap = float(re.findall(r"\d+\.\d+", v8_lines[2])[0])

# 解析YOLOv5指标（需根据val.txt格式调整）
with open("yolov5_metrics.txt", "r") as f:
    v5_lines = f.readlines()
# 示例：val.txt最后几行包含mAP@0.5
v5_map50 = float(re.findall(r"mAP@0.5: (\d+\.\d+)", v5_lines[-2])[0])
v5_car_ap = float(re.findall(r"car: (\d+\.\d+)", v5_lines[-3])[0])
v5_parking_ap = float(re.findall(r"parking_space: (\d+\.\d+)", v5_lines[-3])[0])

# 生成对比表格
print("========== YOLOv5 vs YOLOv8 停车场景对比结果 ==========")

print(f"{'车辆AP@0.5':<15} {v5_car_ap:.4f}     {v8_car_ap:.4f}     {'v8' if v8_car_ap>v5_car_ap else 'v5'}")
print(f"{'车位AP@0.5':<15} {v5_parking_ap:.4f}     {v8_parking_ap:.4f}     {'v8' if v8_parking_ap>v5_parking_ap else 'v5'}")

# 速度对比（需实际测试）
v5_speed = 15  # 示例：实测FPS
v8_speed = 20  # 示例：实测FPS
print(f"{'推理FPS':<15} {v5_speed:<10} {v8_speed:<10} {'v8' if v8_speed>v5_speed else 'v5'}")
```

### 四、停车场景经验结论（提前参考）
| 模型       | 精度（车位检测） | 速度 | 模型大小 | 适配场景                     |
|------------|------------------|------|----------|------------------------------|
| YOLOv5s    | 中等             | 快   | ~14MB    | 边缘设备部署（如停车场闸机） |
| YOLOv8s    | 更高（+5%-10%）| 稍快 | ~21MB    | 服务器/后端部署（核心系统）  |

#### 关键建议：
1. 若你的停车系统部署在**服务器/PC端**：优先选YOLOv8（车位检测精度更高，漏检率低）；
2. 若需部署在**边缘设备（如嵌入式板卡）**：选YOLOv5（模型更小，速度略快，精度损失可接受）；
3. 训练小技巧：
   - 车位标注数据少：用YOLOv8的`pretrained=True`微调，收敛更快；
   - 视频帧训练：加入`perspective=True`（透视变换增强），适配停车场多角度车位；
   - 若训练过拟合：减小`epochs`到30，或增加`augment=True`（数据增强）。
